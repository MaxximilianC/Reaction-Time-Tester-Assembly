There will be two displays. One display has a timer that starts from zero based on the start of the button press and will stop once the button is released
That final time on that display is the reaction time of the user.
The second buzz could be used as a display of increment if the user successfully presses the button within a certain timer

Something that could be done is that we could have an easy level and a hard level where the hard level has a much shorter interaction time.
The decision between easy mode and hard mode can be based on one button input or two.

We will focus on the easy level first, get that working, and decide if we want to add a hard level.

Here is some barebones pseudocode for main concepts (not done & subject to edits & may be wrong):

~~~vector table~~~
.org 0x0000                              ; Reset vector
      jmp       main

.org INT0addr                            ; D2 reaction pressed down button 
      jmp       button_isr

.org OC1Aaddr                            ; timer compare match to random delay
      jmp       t1_ctc_isr

.org INT_VECTORS_SIZE                    ; end vector table 

~~~configuration and turns on RGB pins~~~
main:    
      ldi       r16, high(RAMEND)        ; one time configuration
      out       SPH, r16
      ldi       r16, low(RAMEND)
      out       SPL, r16

      sbi       DDRD, DDD6
      cbi       PORTD, PD6              ; set pin D 6 for red led and turn off
      sbi       DDRD, DDD7
      cbi       PORTD, PD7              ; set pin D 7 for blue led and turn off
      sbi       DDRB, DDB0
      cbi       PORTB, PB0              ; set pin B 0 for green led and turn off

      sbi       EIMSK, INT0             ; reaction button interrupt

      ldi       r16, 0b01               ; INT0 for any logical change (because will react to when button is pressed down to start timer and released to stop timer)
      sts       EIRCA, r16

      call       t1_init                ; enable reaction timer ctc
      sei                               ; enable global interrupts

~~~loop to have the reaction test happen 4 times~~~
      clr       r16                     ; counter set to 0
one_whole_reaction_time_cycle:
      inc       r16                    ; counter++
      cpi       r16, 4                 ; if counter=4 
      breq       end_main              ; end cycle

      cbi        DDRD, DDD2            ; make reaction button an input
      sbi       PORTD, PD2             ; set button to pull-up mode

t1_init:

      clr       r16                    ; start counter at zero
      sts       TCNT1H, r16
      sts       TCNT1L, r16
                                       ; count up to match value (1s = 62499?)
      ldi      r16, high(some kind of number)
      sts      OCR1AH, r16
      ldi      r16, low(some kind of number)
      sts      OCR1AL, r16 

      clr      r16                      ; set mode to ctc
      sts      TCCR1B, r16
                                        ; ctc mode with clock prescalar set to smth (use CS12, CS11, CS10. 1<<WGM12 for ctc mode)
      ldi      r16, (1<<WGM12) | (1<<smth)
      sts      TCCR1B, r16              ; clock is started

      ldi      r16, (1<<OCIE1A)
      sts      TIMSK1, r16              ; enable ctc interrupts

      ret                               ; t1_init

~~~random delay timer turns on when button input is pressed down~~
~~~~~~~timer connected to clock display starts counting when random delay timer reaches the random number~~~~
~~~~~~~~~~~~when button is released timer connect to clock stops (if button released before t.c.t.c reaches end green led lights, if not buzzer sounds)

      rjmp one_whole_reaction_time_cycle

end_main:
      rjmp end_main
